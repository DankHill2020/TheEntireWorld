import jsonlines
from datetime import datetime

class JournalManager:
    def __init__(self, journal_path: str):
        self.journal_path = journal_path

    def log_completion(self, task_name):
        with jsonlines.open(self.journal_path, mode='a') as writer:
            writer.write({'task': task_name, 'status': 'completed', 'timestamp': datetime.now().isoformat()})

    def log_error(self, task_name, error_message):
        with jsonlines.open(self.journal_path, mode='a') as writer:
            writer.write({'task': task_name, 'status': 'error', 'message': error_message, 'timestamp': datetime.now().isoformat()})

    def is_task_completed(self, task_name):
        with jsonlines.open(self.journal_path) as reader:
            for entry in reader:
                if entry['task'] == task_name and entry['status'] == 'completed':
                    return True
        return False

    def get_retry_count(self, task_name):
        count = 0
        with jsonlines.open(self.journal_path) as reader:
            for entry in reader:
                if entry['task'] == task_name and entry['status'] == 'error':
                    count += 1
        return count

from collections import defaultdict

class Task:
    def __init__(self, name):
        self.name = name
        self.dependencies = []

    def add_dependency(self, task):
        self.dependencies.append(task)

    def run(self, observer=None):
        if observer:
            observer.update(f"Running {self.name}")
        # Simulate task execution
        time.sleep(1)
        return True

class DependencyGraph:
    def __init__(self):
        self.tasks = {}
        self.graph = defaultdict(list)

    def add_task(self, task: Task):
        self.tasks[task.name] = task

    def add_dependency(self, task_name, dependency_name):
        if task_name in self.tasks and dependency_name in self.tasks:
            self.graph[dependency_name].append(task_name)
            self.tasks[task_name].add_dependency(self.tasks[dependency_name])

    def topological_sort(self):
        visited = set()
        result = []

        def dfs(node):
            if node not in visited:
                visited.add(node)
                for neighbor in self.graph[node]:
                    dfs(neighbor)
                result.append(node)

        for task in self.tasks.values():
            dfs(task.name)

        return result[::-1]
